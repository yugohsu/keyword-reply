<script>
let PW = '';
let D = null;          // 後台資料
let R = null;          // 正在編輯的規則
let POSTS = [];        // 已載入的貼文
let POSTS_KEY = '';
let TAB = 'rules';
let CHECK = null;      // 權杖檢查結果

const $ = id => document.getElementById(id);

function run(fn, ...args) {
  return new Promise((ok, no) => {
    google.script.run.withSuccessHandler(ok).withFailureHandler(no)[fn](...args);
  });
}
async function call(fn, ...args) {
  $('loading').style.display = 'flex';
  try { return await run(fn, ...args); }
  catch (e) { toast(e.message || String(e), 4500); throw e; }
  finally { $('loading').style.display = 'none'; }
}
function toast(msg, ms) {
  const t = $('toast');
  t.textContent = msg;
  t.style.display = 'block';
  clearTimeout(t._h);
  t._h = setTimeout(() => t.style.display = 'none', ms || 3000);
}
function esc(s) {
  return String(s == null ? '' : s).replace(/[&<>"']/g, c =>
    ({ '&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;', "'":'&#39;' }[c]));
}
function store(k, v) {
  try { if (v === null) localStorage.removeItem(k); else localStorage.setItem(k, v); } catch (e) {}
}
function load(k) {
  try { return localStorage.getItem(k) || ''; } catch (e) { return ''; }
}
function copyText(text) {
  const ta = document.createElement('textarea');
  ta.value = text;
  document.body.appendChild(ta);
  ta.select();
  try { document.execCommand('copy'); toast('已複製'); } catch (e) { toast('請手動複製'); }
  ta.remove();
}
function pageName(id) {
  const p = (D.pages || []).find(x => x.pageId === id);
  return p ? p.name : '(未知粉專)';
}
function showTime(s) {
  return s ? s.replace('T', ' ') : '';
}
function ruleState(r) {
  // 依開始 / 結束時間判斷目前狀態
  if (!r.enabled) return ['off', '已停用'];
  const now = Date.now();
  if (r.startAt && now < new Date(r.startAt).getTime()) return ['warn', '尚未開始'];
  if (r.endAt && now > new Date(r.endAt).getTime()) return ['off', '已結束'];
  return ['ok', '啟用中'];
}

/* ================= 登入 ================= */
async function init() {
  let st;
  try { st = await run('apiStatus'); }
  catch (e) { $('login').innerHTML = '載入失敗：' + esc(e.message); return; }

  if (!st.hasPassword) {
    $('login').innerHTML = `
      <h2>第一次使用，請設定管理密碼</h2>
      <label>管理密碼</label><input id="pw1" type="password">
      <label>再輸入一次</label><input id="pw2" type="password">
      <div class="hint">之後打開這個網頁都要輸入這組密碼，別人拿到網址也不能亂改。</div>
      <br><button class="big" onclick="setFirstPw()">設定密碼</button>`;
    return;
  }
  const saved = load('kw_pw');
  if (saved) {
    try { await doLogin(saved, true); return; } catch (e) {}
  }
  showLogin();
}
function showLogin() {
  $('login').innerHTML = `
    <label>管理密碼</label><input id="pw" type="password" onkeydown="if(event.key==='Enter')loginClick()">
    <br><br><button class="big" onclick="loginClick()">登入</button>`;
}
async function setFirstPw() {
  const a = $('pw1').value, b = $('pw2').value;
  if (a !== b) return toast('兩次密碼不一樣');
  await call('apiSetPassword', a, '');
  await doLogin(a);
}
function loginClick() { doLogin($('pw').value); }
async function doLogin(pw, silent) {
  D = silent ? await run('apiLoad', pw) : await call('apiLoad', pw);
  PW = pw;
  store('kw_pw', pw);
  $('login').classList.add('hide');
  $('app').classList.remove('hide');
  showTab(D.pages.length ? 'rules' : 'pages');
}
function logout() {
  store('kw_pw', null);
  location.reload();
}

/* ================= 分頁 ================= */
function showTab(t) {
  TAB = t;
  document.querySelectorAll('.tab').forEach(b => b.classList.toggle('on', b.dataset.t === t));
  if (t === 'rules') renderRules();
  if (t === 'pages') renderPages();
  if (t === 'logs') renderLogs();
  if (t === 'settings') renderSettings();
  window.scrollTo(0, 0);
}

/* ================= 規則列表 ================= */
function renderRules() {
  if (!D.pages.length) {
    $('main').innerHTML = `<div class="card">還沒有粉專，請先到「粉專」分頁匯入。</div>`;
    return;
  }
  let h = `<button class="big" onclick="openEditor('', false)">新增規則</button>`;
  if (!D.rules.length) h += `<div class="card">目前沒有規則，按上面的按鈕新增第一條。</div>`;
  D.rules.forEach(r => {
    const kw = r.mode === '任何留言' ? '任何內容都回覆' : r.keywords;
    const st = ruleState(r);
    const typeName = { '留言':'貼文留言', '私訊':'私訊', '限動回覆':'限動回覆', '全部':'全部' }[r.type] || r.type;
    let period = '';
    if (r.startAt || r.endAt) period = `期間：${showTime(r.startAt) || '不限'} ～ ${showTime(r.endAt) || '不限'}`;
    h += `<div class="card">
      <div class="kw">${esc(kw)}</div>
      <div>
        <span class="tag ${st[0]}">${st[1]}</span>
        <span class="tag">${esc(pageName(r.pageId))}</span>
        <span class="tag">${esc(r.platform === '全部' ? 'FB + IG' : r.platform)}</span>
        <span class="tag">${esc(typeName)}</span>
        ${(r.type === '留言' || r.type === '全部') ? `<span class="tag">${r.topOnly ? '只回第一層' : '含回覆留言'}</span>` : ''}
        ${r.delayMax > 0 ? `<span class="tag">延遲 ${r.delayMin}～${r.delayMax} 秒</span>` : ''}
        ${(r.links.length || r.quick.length) ? `<span class="tag">按鈕 ${r.links.length + r.quick.length} 個</span>` : ''}
      </div>
      ${period ? `<div class="preview">${esc(period)}</div>` : ''}
      ${(r.type === '留言' || r.type === '全部') ? `<div class="preview">${r.postId ? '指定貼文：' + esc(r.postText.slice(0, 40)) : '全部貼文'}</div>` : ''}
      ${r.publicReply ? `<div class="preview">公開回覆：${esc(r.publicReply.slice(0, 60))}</div>` : ''}
      ${r.dm ? `<div class="preview">私訊：${esc(r.dm.slice(0, 60))}</div>` : ''}
      <div class="row">
        <button class="btn blue" onclick="openEditor('${r.id}', false)">編輯</button>
        <button class="btn blue" onclick="openEditor('${r.id}', true)">複製</button>
        <button class="btn" onclick="toggleRule('${r.id}')">${r.enabled ? '停用' : '啟用'}</button>
        <button class="btn red" onclick="deleteRule('${r.id}')">刪除</button>
      </div>
    </div>`;
  });
  $('main').innerHTML = h;
}
async function toggleRule(id) {
  D = await call('apiToggleRule', PW, id);
  renderRules();
}
async function deleteRule(id) {
  if (!confirm('確定刪除這條規則？')) return;
  D = await call('apiDeleteRule', PW, id);
  renderRules();
  toast('已刪除');
}

/* ================= 規則編輯 ================= */
function openEditor(id, isCopy) {
  const old = id ? D.rules.find(r => r.id === id) : null;
  if (old) {
    R = JSON.parse(JSON.stringify(old));
    if (isCopy) {
      R.id = '';
      R.postId = '';
      R.postText = '';
      R._copy = true;
    }
  } else {
    R = {
      id: '', enabled: true, pageId: D.pages[0].pageId, platform: 'FB', type: '留言',
      postId: '', postText: '', keywords: '', mode: '包含', publicReply: '', dm: '', exclude: '',
      topOnly: true, startAt: '', endAt: '', delayMin: 0, delayMax: 0, links: [], quick: []
    };
  }
  while (R.links.length < 3) R.links.push({ title: '', url: '' });
  R.quickText = (R.quick || []).join(', ');
  POSTS = []; POSTS_KEY = '';
  renderEditor();
  window.scrollTo(0, 0);
  if (isCopy) toast('已複製，請重新選擇貼文並修改內容', 4000);
}
function seg(field, opts) {
  return `<div class="seg">` + opts.map(([v, label]) =>
    `<button class="${R[field] === v ? 'on' : ''}" onclick="setR('${field}','${v}')">${label}</button>`).join('') + `</div>`;
}
function setR(field, v) {
  R[field] = v;
  if (field === 'pageId' || field === 'platform') { R.postId = ''; R.postText = ''; POSTS = []; POSTS_KEY = ''; }
  if (field === 'type' && v !== '留言') { R.postId = ''; R.postText = ''; }
  if (field === 'type' && v === '限動回覆' && R.platform === 'FB') R.platform = 'IG';
  renderEditor();
}
function setBool(field, v) {
  R[field] = v;
  renderEditor();
}
function boolSeg(field, yes, no) {
  return `<div class="seg">
    <button class="${R[field] ? 'on' : ''}" onclick="setBool('${field}', true)">${yes}</button>
    <button class="${!R[field] ? 'on' : ''}" onclick="setBool('${field}', false)">${no}</button></div>`;
}

  let postBlock;
  if (canPickPost) {

      <div class="chosen">${R.postId ? '已選：' + esc(R.postText.slice(0, 60)) : '目前：全部貼文都適用'}</div>

    ${boolSeg('enabled', '啟用', '停用')}

    <h3>測試看看</h3>
    <input id="testText" placeholder="輸入一則假留言，例如：我想去 +1">
    ${isComment ? `<label style="font-size:18px;margin-top:10px"><input type="checkbox" id="testReply" style="width:auto;transform:scale(1.5);margin-right:10px">這是一則回覆留言（不是第一層）</label>` : ''}
    <div class="row"><button class="btn blue" onclick="testRule()">測試</button></div>
    <div id="testResult"></div>

    <br>
    <button class="big" onclick="saveRule()">儲存規則</button>
    <button class="btn" style="width:100%" onclick="showTab('rules')">取消</button>
  </div>`;
  if (canPickPost && POSTS.length) renderPostGrid();
}
function setEndOfMonth() {
  const d = new Date();
  const last = new Date(d.getFullYear(), d.getMonth() + 1, 0);
  const pad = n => String(n).padStart(2, '0');
  R.endAt = `${last.getFullYear()}-${pad(last.getMonth() + 1)}-${pad(last.getDate())}T23:59`;
  renderEditor();
}
async function loadPosts() {
  const key = R.pageId + '|' + R.platform;
  if (POSTS_KEY !== key) {
    POSTS = await call('apiGetPosts', PW, R.pageId, R.platform);
    POSTS_KEY = key;
  }
  if (!POSTS.length) return toast('沒有讀到貼文');
  renderPostGrid();
}
function renderPostGrid() {
  const g = $('postGrid');
  if (!g) return;
  g.innerHTML = `<div class="posts">` + POSTS.map((p, i) => `
    <div class="post ${R.postId === p.id ? 'on' : ''}" onclick="choosePost(${i})">
      ${p.img ? `<img src="${esc(p.img)}" loading="lazy">` : `<div class="noimg"></div>`}
      <div class="t">${esc(p.text.slice(0, 50))}</div>
      <div class="d">${esc(p.time)}</div>
    </div>`).join('') + `</div>`;
}
function choosePost(i) {
  const p = POSTS[i];
  R.postId = p.id;
  R.postText = p.text.slice(0, 80);
  renderEditor();
  toast('已選擇貼文');
}
function ruleForServer() {
  const r = Object.assign({}, R);
  r.links = (R.links || []).filter(l => l.title.trim() && l.url.trim());
  r.quick = String(R.quickText || '').split(/[,，、]/).map(s => s.trim()).filter(Boolean);
  delete r.quickText;
  delete r._copy;
  return r;
}
async function testRule() {
  const t = $('testText').value;
  if (!t) return toast('請先輸入測試文字');
  const isReply = $('testReply') ? $('testReply').checked : false;
  const res = await call('apiTestRule', PW, ruleForServer(), t, isReply);
  $('testResult').innerHTML = `<div class="result">${esc(res)}</div>`;
}
async function saveRule() {
  D = await call('apiSaveRule', PW, ruleForServer());
  toast('已儲存');
  showTab('rules');
}

/* ================= 粉專 ================= */
function renderPages() {
  let h = '';
  if (D.pages.length) {
    h += `<button class="big" onclick="checkPages()">檢查所有粉專權杖</button>`;
  }
  D.pages.forEach(p => {
    const c = CHECK ? CHECK.find(x => x.pageId === p.pageId) : null;
    let status = '';
    if (c) {
      const cls = !c.ok ? 'off' : c.warn ? 'warn' : 'ok';
      status = `<span class="tag ${cls}">${esc(c.ok && !c.warn ? '權杖正常' : c.msg)}</span>`;
    }
    h += `<div class="card">
      <div class="kw">${esc(p.name)}</div>
      ${status}
      <div class="preview">粉專 ID：${esc(p.pageId)}</div>
      <div class="preview">${p.igId ? 'IG：' + esc(p.igName || p.igId) + (p.igOff ? '（已停用，不會回覆 IG）' : '（使用中）') : '沒有連結 IG 專業帳號'}</div>
      <div class="row">
        <button class="btn blue" onclick="resub('${esc(p.pageId)}')">重新訂閱</button>
        ${p.igId ? `<button class="btn" onclick="setIg('${esc(p.pageId)}', ${p.igOff ? 'true' : 'false'})">${p.igOff ? '啟用 IG' : '停用 IG'}</button>` : ''}
        <button class="btn red" onclick="removePage('${esc(p.pageId)}')">移除</button>
      </div>
    </div>`;
  });
  h += `<div class="card">
    <h2>匯入粉專</h2>
    <label>使用者存取權杖</label>
    <textarea id="userToken" placeholder="貼上從 Graph API 測試工具取得的權杖"></textarea>
    <div class="hint">
      取得方式：開發者後台 → 工具 → Graph API 測試工具 → 選你的應用程式 → 勾選權限：
      pages_show_list、pages_read_engagement、pages_manage_metadata、pages_manage_engagement、pages_messaging、
      instagram_basic、instagram_manage_comments、instagram_manage_messages、business_management
      → 產生存取權杖 → 複製貼上。<br>
      先到「設定」填好 App ID 和 App Secret，匯入時會自動換成不會過期的權杖。<br>
      權杖失效時，重新匯入一次就好，規則不會受影響。
    </div>
    <br><button class="big" onclick="importPages()">匯入我管理的所有粉專</button>
    <div id="importResult"></div>
    <details>
      <summary>手動新增（進階）</summary>
      <label>粉專名稱</label><input id="mName">
      <label>粉專 ID</label><input id="mPid">
      <label>粉專存取權杖</label><input id="mTok">
      <label>IG 專業帳號 ID（可不填）</label><input id="mIg">
      <br><br><button class="btn blue" onclick="addPage()">新增</button>
    </details>
  </div>`;
  $('main').innerHTML = h;
}
async function checkPages() {
  CHECK = await call('apiCheckPages', PW);
  renderPages();
  const bad = CHECK.filter(c => !c.ok || c.warn).length;
  toast(bad ? bad + ' 個粉專需要處理' : '全部正常');
}
async function importPages() {
  const tok = $('userToken').value;
  const res = await call('apiImportPages', PW, tok);
  D = res.data;
  CHECK = null;
  renderPages();
  $('importResult').innerHTML = `<div class="result">${esc(res.note + '\n\n' + (res.added.join('\n') || ''))}</div>`;
}
async function addPage() {
  const res = await call('apiAddPage', PW, {
    name: $('mName').value, pageId: $('mPid').value, token: $('mTok').value, igId: $('mIg').value
  });
  D = res.data;
  CHECK = null;
  renderPages();
  toast(res.msg);
}
async function removePage(id) {
  if (!confirm('確定移除這個粉專？（規則不會刪除）')) return;
  D = await call('apiRemovePage', PW, id);
  renderPages();
}
async function setIg(id, enabled) {
  D = await call('apiSetIgEnabled', PW, id, enabled);
  renderPages();
  toast(enabled ? '已啟用 IG' : '已停用 IG，不會回覆這個 IG 帳號');
}
async function resub(id) {
  toast(await call('apiResubscribe', PW, id));
}

/* ================= 紀錄 ================= */
function renderLogs() {
  const s = D.stats;
  let h = `<div class="stats">
    <div class="card">今日成功<b style="color:var(--ok)">${s.ok}</b></div>
    <div class="card">今日失敗<b style="color:var(--bad)">${s.fail}</b></div>
    <div class="card">排隊中<b>${s.waiting}</b></div>
  </div>
  <div class="row" style="margin-bottom:14px">
    <button class="btn blue" onclick="refresh()">重新整理</button>
    ${s.waiting ? `<button class="btn blue" onclick="runQueue()">立即處理排隊</button>` : ''}
  </div>
  <div class="card"><h2>最近紀錄</h2>`;
  if (!D.logs.length) h += `<div class="hint">還沒有紀錄。</div>`;
  else {
    h += `<table><tr><th>時間</th><th>來源</th><th>內容</th><th>結果</th></tr>`;
    D.logs.forEach(l => {
      const color = /失敗/.test(l.result) ? 'var(--bad)' : /排隊|排程|延遲/.test(l.result) ? 'var(--warn)' : 'var(--ok)';
      h += `<tr>
        <td>${esc(l.time)}</td>
        <td>${esc(l.page)}<br>${esc(l.platform)} ${esc(l.type)}<br>${esc(l.who)}</td>
        <td>${esc(l.text)}<br><span style="color:var(--main)">${esc(l.keyword)}</span></td>
        <td style="color:${color}">${esc(l.result)}</td>
      </tr>`;
    });
    h += `</table>`;
  }
  h += `</div>`;
  $('main').innerHTML = h;
}
async function refresh() {
  D = await call('apiLoad', PW);
  showTab(TAB);
  toast('已更新');
}
async function runQueue() {
  D = await call('apiProcessQueueNow', PW);
  renderLogs();
  toast('處理完成');
}
