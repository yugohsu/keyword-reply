/**
 * 小優 FB / IG 關鍵字自動回覆　後台程式（Code.gs）第 2 版
 *
 * 安裝方式：
 *  1. 新建一個 Google 試算表 → 擴充功能 → Apps Script
 *  2. 把這份貼到 Code.gs（整份取代）
 *  3. 左邊「檔案 +」→ HTML → 檔名輸入 Index（大小寫要一樣）→ 貼上 Index.html 內容
 *  4. 部署 → 新增部署作業 → 類型選「網頁應用程式」
 *       執行身分：我　　誰可以存取：所有人
 *  5. 打開部署網址 → 設定管理密碼 → 其餘都在網頁上操作
 *
 * 從第 1 版更新：兩個檔案整份取代 → 部署 → 管理部署作業 → 編輯 → 版本選「新版本」→ 部署
 *   （網址不變，舊規則會自動轉成新格式）
 *
 * 第 2 版新增：
 *  - 可選只回第一層留言
 *  - 規則有效期間（開始 / 結束時間）
 *  - 私訊按鈕（網址按鈕、快速回覆按鈕）
 *  - 回覆前隨機延遲幾秒
 *  - 權杖失效提醒（Email，可加 LINE）
 *  - IG 限時動態回覆
 *  - 複製規則（在網頁上操作）
 */

/* =========================================================
 *  基本工具
 * ========================================================= */
function prop_(k) {
  return (PropertiesService.getScriptProperties().getProperty(k) || '').trim();
}
function setProp_(k, v) {
  PropertiesService.getScriptProperties().setProperty(k, v);
}
function getPages_() {
  try { return JSON.parse(prop_('PAGES') || '[]'); } catch (e) { return []; }
}
function savePages_(pages) {
  setProp_('PAGES', JSON.stringify(pages));
}
function findPage_(id) {
  id = String(id);
  return getPages_().find(function (p) {
    return p.pageId === id || (p.igId && p.igId === id && !p.igOff); // 停用 IG 的粉專不處理 IG 事件
  }) || null;
}
function txt_(v) {
  // 文字前加 ' 讓試算表當純文字存（避免 +1 被當公式、長 ID 被變成科學記號）
  if (v === null || v === undefined || v === '') return '';
  return typeof v === 'string' ? "'" + v : v;
}
function tz_() {
  return Session.getScriptTimeZone();
}
function fmt_(d) {
  try { return Utilities.formatDate(new Date(d), tz_(), 'MM/dd HH:mm'); } catch (e) { return String(d); }
}
function today_() {
  return Utilities.formatDate(new Date(), tz_(), 'yyyy-MM-dd');
}
function parseTime_(s) {
  // '2026-11-30T23:59' → 毫秒（依專案時區）
  s = String(s || '').trim();
  if (!s) return 0;
  try { return Utilities.parseDate(s.replace('T', ' ').slice(0, 16), tz_(), 'yyyy-MM-dd HH:mm').getTime(); }
  catch (e) { return 0; }
}
function ss_() {
  // 有綁定試算表就用它；沒有（獨立的 Apps Script 專案）就自動建立一個，之後都用同一個
  let ss = null;
  try { ss = SpreadsheetApp.getActive(); } catch (e) {}
  if (ss) return ss;
  const id = prop_('SHEET_ID');
  if (id) {
    try { return SpreadsheetApp.openById(id); } catch (e) {}
  }
  ss = SpreadsheetApp.create('關鍵字自動回覆_資料');
  setProp_('SHEET_ID', ss.getId());
  return ss;
}
function sheet_(name, headers) {
  const ss = ss_();
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.getRange(1, 1, 1, headers.length).setValues([headers])
      .setFontWeight('bold').setBackground('#FFE599');
    sh.setFrozenRows(1);
  }
  return sh;
}
function norm_(s) {
  // 全形轉半形 + 小寫
  return String(s || '')
    .replace(/[！-～]/g, function (c) { return String.fromCharCode(c.charCodeAt(0) - 0xFEE0); })
    .replace(/　/g, ' ')
    .toLowerCase();
}
function splitKw_(s) {
  return String(s || '').split(/[,，、\n]/).map(function (k) { return k.trim(); }).filter(String);
}
function pick_(tpl) {
  // 多個版本用一行 --- 隔開，隨機挑一個
  const parts = String(tpl || '').split(/\n\s*---\s*\n/).map(function (t) { return t.trim(); }).filter(String);
  if (!parts.length) return '';
  return parts[Math.floor(Math.random() * parts.length)];
}
function fill_(tpl, name) {
  return String(tpl || '').replace(/\{名字\}/g, name || '');
}
function seen_(id) {
  // 防止 Meta 重送同一個事件造成重複回覆
  const cache = CacheService.getScriptCache();
  if (cache.get('done_' + id)) return true;
  cache.put('done_' + id, '1', 21600);
  return false;
}
function log_(row) {
  sheet_(SH_LOG, LOG_HEADERS).appendRow(row.map(txt_));
}

/* =========================================================
 *  通知（Email 一定會寄，LINE 有設定才送）
 * ========================================================= */
function notify_(title, body) {
  const out = [];
  try {
    MailApp.sendEmail(Session.getEffectiveUser().getEmail(), title, body);
    out.push('Email 已寄出');
  } catch (e) { out.push('Email 失敗：' + e); }

  const lineToken = prop_('LINE_TOKEN'), lineTo = prop_('LINE_TO');
  if (lineToken && lineTo) {
    try {
      const res = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
        method: 'post',
        contentType: 'application/json',
        headers: { Authorization: 'Bearer ' + lineToken },
        payload: JSON.stringify({ to: lineTo, messages: [{ type: 'text', text: (title + '\n\n' + body).slice(0, 4900) }] }),
        muteHttpExceptions: true
      });
      out.push(res.getResponseCode() === 200 ? 'LINE 已送出' : 'LINE 失敗：' + res.getContentText().slice(0, 150));
    } catch (e) { out.push('LINE 失敗：' + e); }
  }
  return out.join('　');
}

function tokenAlert_(page, msg) {
  // 回覆時發現權杖失效：6 小時內同一個粉專只通知一次
  const cache = CacheService.getScriptCache();
  const key = 'tokalert_' + page.pageId;
  if (cache.get(key)) return;
  cache.put(key, '1', 21600);
  notify_('【關鍵字自動回覆】粉專權杖失效：' + page.name,
    page.name + ' 的權杖已失效，自動回覆已停止。\n\n錯誤：' + msg +
    '\n\n請打開管理網頁 →「粉專」分頁 → 重新匯入粉專。');
}

/* =========================================================
 *  Meta Graph API
 * ========================================================= */
function graph_(method, path, params, token) {
  params = params || {};
  if (token) params.access_token = token;
  let url = 'https://graph.facebook.com/' + GRAPH_VER + '/' + path;
  const opt = { method: method, muteHttpExceptions: true };
  if (method === 'get') {
    url += '?' + Object.keys(params).map(function (k) {
      return encodeURIComponent(k) + '=' + encodeURIComponent(params[k]);
    }).join('&');
  } else {
    opt.payload = params;
  }
  try {
    return parse_(UrlFetchApp.fetch(url, opt));
  } catch (e) {
    return fetchErr_(e);
  }
}

function fetchErr_(e) {
  const s = String(e);
  const rl = /too many|頻繁|次數/i.test(s);
  return { ok: false, rateLimited: rl, transient: !rl, msg: '連線逾時或中斷：' + s };
}

// 常見錯誤翻成中文，方便看紀錄
function errText_(code, sub, message) {
  if (code === 1 || code === 2 || code === -1) return 'Meta 暫時忙碌';
  if (code === 190) return '權杖失效，請到「粉專」重新匯入';
  if (code === 10 || code === 200 || code === 230) return '沒有權限，或已超過可以回覆的期限（留言私訊限 7 天內）';
  if (code === 551 || sub === 1545041) return '對方目前無法接收訊息（可能封鎖粉專或關閉訊息）';
  if (sub === 2018278 || sub === 2018108) return '超過可以回覆的期限';
  if (code === 100) return '訊息格式或對象有問題（例如這則留言已經私訊過、留言被刪除）';
  if (RATE_LIMIT_CODES.indexOf(code) >= 0) return '達到 Meta 速度上限';
  return '';
}

// 解析 Meta 的回應
function parse_(res) {
  const code = res.getResponseCode();
  let data = {};
  try { data = JSON.parse(res.getContentText()); } catch (e) {}
  if (code === 200 && !data.error) return { ok: true, data: data };
  const er = data.error || {};
  const rl = RATE_LIMIT_CODES.indexOf(er.code) >= 0 || er.error_subcode === 2534040 ||
    /rate limit|limit reached|too many/i.test(er.message || '');
  // Meta 那邊暫時出錯（不是我們的問題），重送通常就會成功
  const transient = !rl && (er.is_transient === true || er.code === 1 || er.code === 2 || er.code === -1 ||
    (!er.code && code >= 500));
  const zh = errText_(er.code, er.error_subcode, er.message);
  return {
    ok: false,
    rateLimited: rl,
    transient: transient,
    tokenBad: er.code === 190,
    msg: '(' + (er.code || code) + ') ' + (zh ? zh + '｜' : '') + (er.message || res.getContentText().slice(0, 200))
  };
}

// 把 {text, links, quick} 轉成 Messenger / IG 的訊息格式
function toMessengerMsg_(msg, plain) {
  const text = String(msg.text || '');
  const links = (msg.links || []).filter(function (l) { return l && l.title && l.url; }).slice(0, 3);
  const quick = (msg.quick || []).filter(String).slice(0, 13);
  let m;
  if (plain) {
    // 按鈕格式被拒絕時的備案：網址直接接在文字後面
    const extra = links.map(function (l) { return l.title + '：' + l.url; }).join('\n');
    m = { text: (text + (extra ? '\n\n' + extra : '')).slice(0, 2000) };
    return m;
  }
  if (links.length) {
    m = {
      attachment: {
        type: 'template',
        payload: {
          template_type: 'button',
          text: (text || '請點選下方按鈕').slice(0, 640),
          buttons: links.map(function (l) {
            return { type: 'web_url', url: l.url, title: String(l.title).slice(0, 20) };
          })
        }
      }
    };
  } else {
    m = { text: text.slice(0, 2000) };
  }
  if (quick.length) {
    m.quick_replies = quick.map(function (q) {
      return { content_type: 'text', title: String(q).slice(0, 20), payload: String(q).slice(0, 1000) };
    });
  }
  return m;
}

// 組出一個要送給 Meta 的請求（可以多個一起送）
function sendReq_(page, platform, action, target, msg, plain) {
  let path, payload;
  if (action === '公開') {
    path = platform === 'FB' ? target + '/comments' : target + '/replies';
    payload = { message: msg.text };
  } else {
    const recipient = action === '私訊留言' ? { comment_id: target } : { id: target };
    path = page.pageId + '/messages';
    payload = { recipient: JSON.stringify(recipient), message: JSON.stringify(toMessengerMsg_(msg, plain)) };
    if (action !== '私訊留言') payload.messaging_type = 'RESPONSE';
  }
  payload.access_token = page.token;
  return {
    url: 'https://graph.facebook.com/' + GRAPH_VER + '/' + path,
    method: 'post',
    payload: payload,
    muteHttpExceptions: true
  };
}

// 送出單一請求（UrlFetchApp.fetch 要「網址 + 設定」分開給）
function fetchOne_(req) {
  const opt = {};
  Object.keys(req).forEach(function (k) { if (k !== 'url') opt[k] = req[k]; });
  return UrlFetchApp.fetch(req.url, opt);
}

function hasButtons_(msg) {
  return !!((msg.links && msg.links.length) || (msg.quick && msg.quick.length));
}

// 按鈕被拒絕（例如 IG 不支援某種格式）→ 改送純文字
function retryPlain_(page, platform, action, target, msg, r) {
  if (r.ok || action === '公開' || !hasButtons_(msg) || r.rateLimited || r.tokenBad || r.transient) return r;
  let r2;
  try { r2 = parse_(fetchOne_(sendReq_(page, platform, action, target, msg, true))); }
  catch (e) { return r; }
  if (r2.ok) { r2.note = '（按鈕不支援，已改送文字）'; return r2; }
  return r;
}

function send_(page, platform, action, target, msg) {
  let r;
  try { r = parse_(fetchOne_(sendReq_(page, platform, action, target, msg, false))); }
  catch (e) { r = fetchErr_(e); }
  return retryPlain_(page, platform, action, target, msg, r);
}

function subscribe_(page) {
  return graph_('post', page.pageId + '/subscribed_apps',
    { subscribed_fields: 'feed,messages' }, page.token);
}

/* =========================================================
 *  規則
 * ========================================================= */
function normRule_(r) {
  let btn = { links: [], quick: [] };
  if (r.links || r.quick) {
    btn.links = r.links || [];
    btn.quick = Array.isArray(r.quick) ? r.quick : splitKw_(r.quick);
  } else if (r.buttons) {
    try { btn = JSON.parse(r.buttons); } catch (e) {}
  }
  const links = (btn.links || []).map(function (l) {
    return { title: String(l.title || '').trim(), url: String(l.url || '').trim() };
  }).filter(function (l) { return l.title && l.url; }).slice(0, 3);
  const quick = (btn.quick || []).map(function (q) { return String(q).trim(); }).filter(String).slice(0, 13);

  let dmin = Math.max(0, Number(r.delayMin) || 0);
  let dmax = Math.max(0, Number(r.delayMax) || 0);
  if (dmax < dmin) dmax = dmin;

  const o = {
    id: String(r.id || ''),
    enabled: r.enabled === true || String(r.enabled).toUpperCase() === 'Y',
    pageId: String(r.pageId || ''),
    platform: String(r.platform || '全部'),
    type: String(r.type || '留言'),
    postId: String(r.postId || ''),
    postText: String(r.postText || ''),
    keywords: String(r.keywords || ''),
    mode: String(r.mode || '包含'),
    dmMode: String(r.dmMode || r.mode || '包含'),
    publicReply: String(r.publicReply || ''),
    dm: String(r.dm || ''),
    exclude: String(r.exclude || ''),
    topOnly: r.topOnly === true || String(r.topOnly).toUpperCase() === 'Y',
    startAt: String(r.startAt || '').slice(0, 16),
    endAt: String(r.endAt || '').slice(0, 16),
    delayMin: dmin,
    delayMax: dmax,
    links: links,
    quick: quick,
    observe: r.observe === true || String(r.observe).toUpperCase() === 'Y',
    // 沒設定過的舊規則，預設也是「同一人只回一次」
    once: r.once === undefined || r.once === '' || r.once === null ? true :
      (r.once === true || String(r.once).toUpperCase() === 'Y')
  };
  o.startMs = parseTime_(o.startAt);
  o.endMs = parseTime_(o.endAt);
  return o;
}

function migrateRules_() {
  // 第 1 版的規則表（13 欄）轉成第 2 版（19 欄）
  const sh = ss_().getSheetByName(SH_RULES);
  if (!sh) return;
  const lastCol = sh.getLastColumn();
  const head = sh.getRange(1, 1, 1, Math.max(lastCol, 1)).getValues()[0];
  if (head.length >= RULE_HEADERS.length && head[RULE_HEADERS.length - 1] === '建立時間') return;
  if (head[12] !== '建立時間') return;

  const n = sh.getLastRow();
  const rows = n >= 2 ? sh.getRange(2, 1, n - 1, 13).getValues() : [];
  sh.clear();
  sh.getRange(1, 1, 1, RULE_HEADERS.length).setValues([RULE_HEADERS])
    .setFontWeight('bold').setBackground('#FFE599');
  sh.setFrozenRows(1);
  if (rows.length) {
    const out = rows.map(function (r) {
      return [txt_(String(r[0])), r[1], txt_(String(r[2])), r[3], r[4], txt_(String(r[5])), txt_(String(r[6])),
        txt_(String(r[7])), r[8], txt_(String(r[9])), txt_(String(r[10])), txt_(String(r[11])),
        'N', '', '', 0, 0, '', r[12], 'N'];
    });
    sh.getRange(2, 1, out.length, RULE_HEADERS.length).setValues(out);
  }
}

function migrateQueue_() {

  const ss = ss_();

function getRules_() {
  const sh = sheet_(SH_RULES, RULE_HEADERS);
  const n = sh.getLastRow();
  if (n < 2) return [];
  const rows = sh.getRange(2, 1, n - 1, RULE_HEADERS.length).getValues();
  const list = rows.filter(function (r) { return r[0]; }).map(function (r) {
    const o = normRule_({
      id: r[0], enabled: r[1], pageId: r[2], platform: r[3], type: r[4], postId: r[5],
      postText: r[6], keywords: r[7], mode: r[8], publicReply: r[9], dm: r[10], exclude: r[11],
      topOnly: r[12], startAt: r[13] instanceof Date ? Utilities.formatDate(r[13], tz_(), "yyyy-MM-dd'T'HH:mm") : r[13],
      endAt: r[14] instanceof Date ? Utilities.formatDate(r[14], tz_(), "yyyy-MM-dd'T'HH:mm") : r[14],
      delayMin: r[15], delayMax: r[16], buttons: r[17], observe: r[19], dmMode: r[20], once: r[21]
    });
    o.created = r[18] ? fmt_(r[18]) : '';
    return o;
  });
  // 指定貼文的規則優先比對
  list.sort(function (a, b) { return (b.postId ? 1 : 0) - (a.postId ? 1 : 0); });
  return list;
}

function getRulesCached_() {
  const cache = CacheService.getScriptCache();
  const c = cache.get('rules');
  if (c) { try { return JSON.parse(c); } catch (e) {} }
  const rules = getRules_();
  try { cache.put('rules', JSON.stringify(rules), 21600); } catch (e) {} // 6 小時，改規則時會自動更新
  return rules;
}

function refreshRulesCache_() {
  const cache = CacheService.getScriptCache();
  cache.remove('rules');
  getRulesCached_();
}

// 類型可複選：'留言,私訊,限動回覆'（舊資料的 '全部' 等於三種都要）
function typesOf_(t) {
  t = String(t || '留言').trim();
  if (t === '全部') return ['留言', '私訊', '限動回覆'];
  return t.split(/[,，、]/).map(function (x) { return x.trim(); }).filter(String);
}

// 指定貼文可複選，用逗號存多個貼文 ID
function postsOf_(s) {
  return String(s || '').split(',').map(function (x) { return x.trim(); }).filter(String);
}

function samePost_(a, b) {
  a = String(a || ''); b = String(b || '');
  if (!a || !b) return false;
  return a === b || a.split('_').pop() === b.split('_').pop();
}

/**
 * ev = { pageId, platform, type('留言'|'私訊'|'限動回覆'), postId, text, isReply }
 * ignoreTime = true 時不檢查有效期間（測試用）
 */
function matchRule_(rules, ev, ignoreTime) {
  const t = norm_(ev.text);
  const now = Date.now();
  let anyRule = null;
  for (let i = 0; i < rules.length; i++) {
    const r = rules[i];
    if (!r.enabled) continue;
    if (r.pageId && r.pageId !== ev.pageId) continue;
    if (r.platform !== '全部' && r.platform !== ev.platform) continue;
    if (typesOf_(r.type).indexOf(ev.type) < 0) continue;
    // 指定貼文只限制「貼文留言」；私訊、限動回覆不受影響
    if (r.postId && ev.type === '留言' &&
        !postsOf_(r.postId).some(function (id) { return samePost_(id, ev.postId); })) continue;
    if (ev.type === '留言' && r.topOnly && ev.isReply) continue;
    if (!ignoreTime) {
      if (r.startMs && now < r.startMs) continue;
      if (r.endMs && now > r.endMs) continue;
    }

    const ex = splitKw_(r.exclude);
    if (ex.some(function (k) { return t.indexOf(norm_(k)) >= 0; })) continue;

    // 留言用「留言比對方式」，私訊和限動回覆用「私訊比對方式」
    const md = ev.type === '留言' ? r.mode : (r.dmMode || r.mode);
    // 「任何內容」的規則先記下來，最後才用：有關鍵字符合的規則優先
    if (md === '任何留言') { if (!anyRule) anyRule = Object.assign({ hit: '(任何內容)' }, r); continue; }

    const kws = splitKw_(r.keywords);
    for (let j = 0; j < kws.length; j++) {
      const nk = norm_(kws[j]).trim();
      if (!nk) continue;
      const ok = md === '完全相同' ? t.trim() === nk : t.indexOf(nk) >= 0;
      if (ok) return Object.assign({ hit: kws[j] }, r);
    }
  }
  return anyRule;
}

// 同一人只回一次：同一條規則、同一個人（留言再加上同一篇貼文），6 小時內只回第一次
function onceKey_(rule, who, postId) {
  return 'once_' + (rule.id || 'x') + '_' + who + (postId ? '_' + postId : '');
}
function repliedBefore_(rule, who, postId) {
  if (!rule.once || !who) return false;
  return CacheService.getScriptCache().get(onceKey_(rule, who, postId)) !== null;
}
function unmark_(rule, who, postId) {
  if (rule.once && who) CacheService.getScriptCache().remove(onceKey_(rule, who, postId));
}
function markReplied_(rule, who, postId, info) {
  if (!rule.once || !who || isObserve_() || info.observe) return;
  CacheService.getScriptCache().put(onceKey_(rule, who, postId), '1', 21600);
}

/* ---------- 觀察模式 ---------- */
function isObserve_() { return prop_('OBSERVE') === '1'; }
function isLogAll_() { return prop_('LOG_ALL') === '1'; }

// 沒觸發時，找出原因（只在「記錄所有訊息」開啟時使用）
function missReason_(rules, ev) {
  if (!rules.some(function (r) { return r.enabled && (!r.pageId || r.pageId === ev.pageId); })) return '這個粉專沒有啟用中的規則';
  if (matchRule_(rules, ev, true)) return '有符合的規則，但不在有效期間';
  if (ev.type === '留言' && ev.isReply && matchRule_(rules, Object.assign({}, ev, { isReply: false }))) return '這是回覆留言，規則設定只回第一層';
  if (ev.type === '留言' && matchRule_(rules.map(function (r) { return Object.assign({}, r, { postId: '' }); }), ev)) return '不是規則指定的貼文';
  return '沒有符合的關鍵字';
}

// 觀察模式：不送出，只記錄會回什麼，並確認權杖讀得到這則留言
function observe_(info, target, jobs) {
  const parts = [isObserve_() ? '觀察模式（沒有送出）' : '只觀察（沒有送出）'];
  jobs.forEach(function (j) {
    const msg = j[1];
    let t = (j[0] === '公開' ? '會公開回覆：' : '會私訊：') + String(msg.text || '').replace(/\s+/g, ' ').slice(0, 60);
    const nb = (msg.links ? msg.links.length : 0) + (msg.quick ? msg.quick.length : 0);
    if (j[0] !== '公開' && nb) t += '［按鈕 ' + nb + ' 個］';
    parts.push(t);
  });
  if (jobs.some(function (j) { return j[0] !== '私訊'; })) {
    const r = graph_('get', target, { fields: 'id' }, info.page.token);
    parts.push(r.ok ? '檢查：權杖可以讀取這則留言' : '檢查失敗：' + r.msg);
    if (r.tokenBad) tokenAlert_(info.page, r.msg);
  }
  return parts.join(' / ');
}

// 私訊沒有附名字，要另外查（查過的記 6 小時）
function userName_(page, platform, sid) {
  const cache = CacheService.getScriptCache();
  const key = 'name_' + sid;
  const c = cache.get(key);
  if (c !== null) return c;
  const r = graph_('get', sid, { fields: platform === 'IG' ? 'username,name' : 'name' }, page.token);
  const name = r.ok ? String(r.data.name || r.data.username || '') : '';
  cache.put(key, name, 21600);
  return name;
}

function randDelay_(rule) {
  const a = Number(rule.delayMin) || 0, b = Number(rule.delayMax) || 0;
  if (b <= 0) return 0;
  return Math.round(a + Math.random() * (b - a));
}

/* =========================================================
 *  Webhook：Meta 驗證（GET）／ 管理網頁

function doGet(e) {
  const p = (e && e.parameter) || {};
  if (p['hub.mode'] === 'subscribe') {

  }

    .setTitle('關鍵字自動回覆')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); // 允許放進 GitHub Pages 網頁裡
}

/* =========================================================
 *  Webhook：收到留言 / 私訊（POST）
 * ========================================================= */
function doPost(e) {
  const ctx = { slept: 0 };
  try {
    const body = JSON.parse(e.postData.contents);
    const platform = body.object === 'instagram' ? 'IG' : 'FB';
    const rules = getRulesCached_();
    (body.entry || []).forEach(function (entry) {
      const page = findPage_(entry.id);
      if (!page) return;
      (entry.changes || []).forEach(function (ch) { onComment_(page, platform, ch, rules, ctx); });
      (entry.messaging || []).forEach(function (m) { onMessage_(page, platform, m, rules, ctx); });
    });
  } catch (err) {
    log_([new Date(), '', '', '錯誤', '', '', '', '', String(err)]);
  }
  return ContentService.createTextOutput('OK');
}

function onComment_(page, platform, ch, rules, ctx) {
  const v = ch.value || {};
  let cid, text, fromId, name, postId, isReply;
  if (platform === 'FB') {
    if (ch.field !== 'feed' || v.item !== 'comment' || v.verb !== 'add') return;
    cid = v.comment_id;
    text = v.message || '';
    fromId = v.from && v.from.id;
    name = (v.from && v.from.name) || '';
    postId = v.post_id || '';
    isReply = !!(v.parent_id && v.parent_id !== v.post_id);
    if (fromId === page.pageId) return;
  } else {
    if (ch.field !== 'comments') return;
    cid = v.id;
    text = v.text || '';
    fromId = v.from && v.from.id;
    name = (v.from && v.from.username) || '';
    postId = (v.media && v.media.id) || '';
    isReply = !!v.parent_id;
    if (fromId === page.igId) return;
  }
  if (!cid || seen_(cid)) return;

  const ev = { pageId: page.pageId, platform: platform, type: '留言', postId: postId, text: text, isReply: isReply };
  const rule = matchRule_(rules, ev);
  if (!rule) {
    if (isLogAll_()) log_([new Date(), page.name, platform, isReply ? '留言(回覆)' : '留言', name, text, postId, '',
      '未觸發：' + missReason_(rules, ev), txt_(String(cid))]);
    return;
  }

  const info = { page: page, platform: platform, who: name, text: text, keyword: rule.hit, observe: rule.observe };
  if (repliedBefore_(rule, fromId, postId)) {
    log_([new Date(), page.name, platform, isReply ? '留言(回覆)' : '留言', name, text, postId, rule.hit,
      '未觸發：同一人在這篇貼文 6 小時內已經回過（規則設定同一人只回一次）', txt_(String(cid))]);
    return;
  }
  markReplied_(rule, fromId, postId, info);
  const jobs = [];
  if (rule.publicReply) jobs.push(['公開', { text: fill_(pick_(rule.publicReply), name) }]);
  if (rule.dm) jobs.push(['私訊留言', { text: fill_(pick_(rule.dm), name), links: rule.links, quick: rule.quick }]);

  const results = runJobs_(info, cid, jobs, randDelay_(rule), ctx);
  if (/失敗/.test(results) && !/成功|排隊|排程|重送/.test(results)) unmark_(rule, fromId, postId);
  log_([new Date(), page.name, platform, isReply ? '留言(回覆)' : '留言', name, text, postId, rule.hit, results, txt_(String(cid))]);
}

function onMessage_(page, platform, m, rules, ctx) {
  if (!m.message || m.message.is_echo) return;
  const mid = m.message.mid;
  if (!mid || seen_(mid)) return;
  const sid = m.sender && m.sender.id;
  if (!sid || sid === page.pageId || sid === page.igId) return;

  const text = m.message.text || '';
  const isStory = !!(m.message.reply_to && m.message.reply_to.story);
  const base = { pageId: page.pageId, platform: platform, postId: '', text: text, isReply: false };

  let type = isStory ? '限動回覆' : '私訊';
  let rule = matchRule_(rules, Object.assign({ type: type }, base));
  if (!rule && isStory) rule = matchRule_(rules, Object.assign({ type: '私訊' }, base)); // 沒有限動規則就用私訊規則
  if (!rule || !rule.dm) {
    if (isLogAll_()) log_([new Date(), page.name, platform, type, userName_(page, platform, sid) || sid, text || '(貼圖或圖片)', '', '',
      '未觸發：' + (rule ? '規則沒有填私訊內容' : missReason_(rules, Object.assign({ type: type }, base)))]);
    return;
  }

  const name = userName_(page, platform, sid);
  const info = { page: page, platform: platform, who: name || sid, text: text || '(貼圖或圖片)', keyword: rule.hit, observe: rule.observe };
  if (repliedBefore_(rule, sid, '')) {
    log_([new Date(), page.name, platform, type, info.who, info.text, '', rule.hit,
      '未觸發：同一人 6 小時內已經回過（規則設定同一人只回一次）']);
    return;
  }
  markReplied_(rule, sid, '', info);
  const jobs = [['私訊', { text: fill_(pick_(rule.dm), name), links: rule.links, quick: rule.quick }]];
  const results = runJobs_(info, sid, jobs, randDelay_(rule), ctx);
  if (/失敗/.test(results) && !/成功|排隊|排程|重送/.test(results)) unmark_(rule, sid, '');
  log_([new Date(), page.name, platform, type, info.who, info.text, '', rule.hit, results]);
}

function runJobs_(info, target, jobs, delay, ctx) {
  if (isObserve_() || info.observe) return observe_(info, target, jobs);
  const out = [];
  if (delay > 0) {
    if (delay <= INLINE_DELAY_MAX && ctx.slept + delay <= INLINE_DELAY_TOTAL) {
      Utilities.sleep(delay * 1000);
      ctx.slept += delay;
      out.push('等待' + delay + '秒');
    } else {
      const due = new Date(Date.now() + delay * 1000);
      jobs.forEach(function (j) { enqueue_(info, j[0], target, j[1], due); });
      return '延遲 ' + delay + ' 秒，已排程';
    }
  }

  const cache = CacheService.getScriptCache();
  if (cache.get('rl_' + info.page.pageId)) {
    jobs.forEach(function (j) { enqueue_(info, j[0], target, j[1], new Date()); });
    return out.concat(['排隊中']).join(' / ');
  }

  // 公開回覆和私訊同時送出
  let results;
  try {
    const resps = UrlFetchApp.fetchAll(jobs.map(function (j) {
      return sendReq_(info.page, info.platform, j[0], target, j[1], false);
    }));
    results = resps.map(parse_);
  } catch (e) {
    results = jobs.map(function () { return fetchErr_(e); });
  }

  jobs.forEach(function (j, i) {
    let r = retryPlain_(info.page, info.platform, j[0], target, j[1], results[i]);
    // Meta 暫時出錯 → 等 1.5 秒馬上重送一次
    if (!r.ok && r.transient) {
      Utilities.sleep(1500);
      r = send_(info.page, info.platform, j[0], target, j[1]);
      if (r.ok) r.note = (r.note || '') + '（第一次 Meta 忙碌，重送成功）';
    }
    out.push((j[0] === '公開' ? '公開:' : '私訊:') + handleResult_(info, j[0], target, j[1], r));
  });
  return out.join(' / ');
}

function handleResult_(info, action, target, msg, r) {
  if (r.ok) return '成功' + (r.note || '');
  if (r.rateLimited) {
    CacheService.getScriptCache().put('rl_' + info.page.pageId, '1', 600); // 暫停 10 分鐘，期間全部改排隊
    enqueue_(info, action, target, msg, new Date());
    return '達上限，排隊中';
  }
  if (r.transient) {
    // 還是失敗 → 1 分鐘後由排隊自動再送（最多再試 3 次）
    enqueue_(info, action, target, msg, new Date(Date.now() + 60000));
    return 'Meta 暫時忙碌，1 分鐘後自動重送';
  }
  if (r.tokenBad) tokenAlert_(info.page, r.msg);
  // 記到排隊表（狀態「失敗」），之後可以在「紀錄」分頁按「重送」
  try { enqueue_(info, action, target, msg, new Date(), r.msg); } catch (e) {}
  return '失敗 ' + r.msg;
}

/* =========================================================
 *  排隊（延遲回覆、超過上限補發）每分鐘自動執行
 * ========================================================= */
function enqueue_(info, action, target, msg, due, failMsg) {
  sheet_(SH_QUEUE, QUEUE_HEADERS).appendRow([
    new Date(), due || new Date(), txt_(info.page.pageId), info.platform, action, txt_(String(target)),
    txt_(JSON.stringify(msg)), txt_(String(info.who || '')), txt_(String(info.text || '')),
    txt_(String(info.keyword || '')), failMsg ? '失敗' : '等待', failMsg ? txt_(failMsg) : '', failMsg ? 1 : 0
  ]);
  if (!failMsg) setProp_('QUEUE_HAS', '1');
}

// 找出同一個人、同一段內容的失敗紀錄，改成「已重送成功」
function clearFailLog_(pageName, who, text) {
  const sh = sheet_(SH_LOG, LOG_HEADERS);
  const n = sh.getLastRow();
  if (n < 2) return;
  const from = Math.max(2, n - 2999);
  const rng = sh.getRange(from, 1, n - from + 1, LOG_HEADERS.length);
  const rows = rng.getValues();
  const when = Utilities.formatDate(new Date(), tz_(), 'MM/dd HH:mm');
  rows.forEach(function (r, i) {
    const res = String(r[8]);
    if (String(r[1]) !== pageName || String(r[4]) !== who || String(r[5]) !== text) return;
    if (!/失敗/.test(res) || /已重送成功/.test(res)) return;
    sh.getRange(from + i, 9).setValue('已重送成功（' + when + '）｜原本：' + res);
  });
}

// 一次性：之前已經重送成功的，把原本的失敗紀錄也改掉
function fixResentLogs_() {
  const sh = sheet_(SH_QUEUE, QUEUE_HEADERS);
  const n = sh.getLastRow();
  if (n < 2) return;
  const pages = getPages_();
  sh.getRange(2, 1, n - 1, QUEUE_HEADERS.length).getValues().forEach(function (r) {
    if (r[Q.status] !== '完成') return;
    const pg = pages.find(function (p) { return p.pageId === String(r[Q.pageId]); });
    if (pg) clearFailLog_(pg.name, String(r[Q.who]), String(r[Q.text]));
  });
}

function processQueue() {
  if (prop_('QUEUE_HAS') !== '1') return;
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(1000)) return;
  try {
    const sh = sheet_(SH_QUEUE, QUEUE_HEADERS);
    const n = sh.getLastRow();
    if (n < 2) { setProp_('QUEUE_HAS', '0'); return; }
    const data = sh.getRange(2, 1, n - 1, QUEUE_HEADERS.length).getValues();
    const pages = getPages_();
    const cache = CacheService.getScriptCache();
    const start = Date.now();
    let sent = 0, stillWaiting = 0, firstWaiting = -1;

    for (let i = 0; i < data.length; i++) {
      const r = data[i];
      if (r[Q.status] !== '等待') continue;
      const pid = String(r[Q.pageId]);
      const due = r[Q.due] instanceof Date ? r[Q.due].getTime() : 0;
      if (due > Date.now() || sent >= 150 || Date.now() - start > 240000 || cache.get('rl_' + pid)) {
        stillWaiting++;
        if (firstWaiting < 0) firstWaiting = i;
        continue;
      }
      const page = pages.find(function (p) { return p.pageId === pid; });
      const tries = Number(r[Q.tries] || 0) + 1;
      let status, result, msg;
      try { msg = JSON.parse(String(r[Q.msg])); } catch (e) { msg = { text: String(r[Q.msg]) }; }

      if (!page) {
        status = '失敗'; result = '找不到粉專';
      } else {
        const res = send_(page, String(r[Q.platform]), String(r[Q.action]), String(r[Q.target]), msg);
        if (res.ok) {
          status = '完成'; result = '成功' + (res.note || ''); sent++;
          // 重送成功 → 把原本那筆失敗紀錄改成「已重送成功」，不再算失敗
          if (tries > 1 || String(r[Q.result]) === '手動重送') {
            try { clearFailLog_(page.name, String(r[Q.who]), String(r[Q.text])); } catch (e) {}
          }
        }
        else if (res.rateLimited) { cache.put('rl_' + pid, '1', 600); status = '等待'; result = '達上限，稍後再試'; }
        else {
          if (res.tokenBad) tokenAlert_(page, res.msg);
          status = tries >= (res.transient ? 5 : 3) || res.tokenBad ? '失敗' : '等待';
          result = res.msg;
        }
      }
      sh.getRange(i + 2, Q.status + 1, 1, 3).setValues([[status, txt_(result), tries]]);
      if (status === '等待') {
        stillWaiting++;
        if (firstWaiting < 0) firstWaiting = i;
      } else {
        log_([new Date(), page ? page.name : pid, String(r[Q.platform]), '排隊送出', String(r[Q.who]),
          String(r[Q.text]), '', String(r[Q.keyword]), String(r[Q.action]) + ':' + (status === '失敗' ? '失敗 ' : '') + result]);
      }
    }
    if (stillWaiting === 0) setProp_('QUEUE_HAS', '0');

    // 清理：前面已經處理完的列超過 500 列就刪掉，避免表格越來越大
    const doneTop = firstWaiting < 0 ? data.length : firstWaiting;
    if (doneTop >= 500) sh.deleteRows(2, doneTop);
  } finally {
    lock.releaseLock();
  }
}

/* =========================================================
 *  權杖檢查（每天早上 9 點自動執行）
 * ========================================================= */
function checkPages_() {
  const appId = prop_('APP_ID'), secret = prop_('APP_SECRET');
  return getPages_().map(function (p) {
    const me = graph_('get', 'me', { fields: 'id,name' }, p.token);
    if (!me.ok) return { pageId: p.pageId, name: p.name, ok: false, msg: '權杖失效 ' + me.msg };

    const warns = [];
    if (appId && secret) {
      const d = graph_('get', 'debug_token', { input_token: p.token }, appId + '|' + secret);
      if (d.ok) {
        const x = d.data.data || {};
        if (x.is_valid === false) return { pageId: p.pageId, name: p.name, ok: false, msg: '權杖失效' };
        if (x.expires_at && x.expires_at > 0) {
          const days = Math.floor((x.expires_at * 1000 - Date.now()) / 86400000);
          if (days <= 7) warns.push('權杖將在 ' + Math.max(days, 0) + ' 天後到期');
        }
        if (x.data_access_expires_at && x.data_access_expires_at > 0) {
          const days2 = Math.floor((x.data_access_expires_at * 1000 - Date.now()) / 86400000);
          if (days2 <= 7) warns.push('資料存取權將在 ' + Math.max(days2, 0) + ' 天後到期');
        }
      }
    }
    const s = graph_('get', p.pageId + '/subscribed_apps', {}, p.token);
    if (s.ok && !(s.data.data || []).length) warns.push('粉專訂閱已中斷，請按「重新訂閱」');

    return { pageId: p.pageId, name: p.name, ok: true, warn: warns.length > 0, msg: warns.join('，') || '正常' };
  });
}

function checkTokens() {
  const res = checkPages_();
  const bad = res.filter(function (r) { return !r.ok || r.warn; });
  if (!bad.length) return;
  notify_('【關鍵字自動回覆】粉專權杖提醒',
    bad.map(function (r) { return r.name + '：' + r.msg; }).join('\n') +
    '\n\n請打開管理網頁 →「粉專」分頁 → 重新匯入粉專或重新訂閱。');
}

/* =========================================================
 *  觸發器與初始化
 * ========================================================= */
function ensureTriggers_() {
  const TRIGGER_VER = '2';
  const triggers = ScriptApp.getProjectTriggers();
  if (prop_('TRIGGER_VER') !== TRIGGER_VER) {
    triggers.forEach(function (t) {
      const f = t.getHandlerFunction();
      if (f === 'processQueue' || f === 'checkTokens') ScriptApp.deleteTrigger(t);
    });
    ScriptApp.newTrigger('processQueue').timeBased().everyMinutes(1).create();
    ScriptApp.newTrigger('checkTokens').timeBased().atHour(9).everyDays(1).create();
    setProp_('TRIGGER_VER', TRIGGER_VER);
    return;
  }
  const names = triggers.map(function (t) { return t.getHandlerFunction(); });
  if (names.indexOf('processQueue') < 0) ScriptApp.newTrigger('processQueue').timeBased().everyMinutes(1).create();
  if (names.indexOf('checkTokens') < 0) ScriptApp.newTrigger('checkTokens').timeBased().atHour(9).everyDays(1).create();
}

function ensureSetup_() {
  if (prop_('SETUP_VER') !== '2') {
    migrateRules_();
    migrateQueue_();
    setProp_('SETUP_VER', '2');
    refreshRulesCache_();
  }
  const rsh = sheet_(SH_RULES, RULE_HEADERS);
  for (let c = 20; c <= RULE_HEADERS.length; c++) {
    if (rsh.getRange(1, c).getValue() !== RULE_HEADERS[c - 1]) {
      rsh.getRange(1, c).setValue(RULE_HEADERS[c - 1]).setFontWeight('bold').setBackground('#FFE599');
    }
  }
  const lsh = sheet_(SH_LOG, LOG_HEADERS);
  if (lsh.getRange(1, LOG_HEADERS.length).getValue() !== LOG_HEADERS[LOG_HEADERS.length - 1]) {
    lsh.getRange(1, LOG_HEADERS.length).setValue(LOG_HEADERS[LOG_HEADERS.length - 1]).setFontWeight('bold');
  }
  sheet_(SH_QUEUE, QUEUE_HEADERS);
  if (prop_('FIX_RESENT') !== '1') {
    setProp_('FIX_RESENT', '1');
    try { fixResentLogs_(); } catch (e) {}
  }

function loadData_() {
  const pages = getPages_().map(function (p) {
    return { pageId: p.pageId, name: p.name, igId: p.igId || '', igName: p.igName || '', igOff: !!p.igOff };
  });

  // 今日統計 + 最近紀錄
  const lsh = sheet_(SH_LOG, LOG_HEADERS);
  const ln = lsh.getLastRow();
  let logs = [], ok = 0, fail = 0;
  if (ln >= 2) {
    const from = Math.max(2, ln - 2999);
    const rows = lsh.getRange(from, 1, ln - from + 1, LOG_HEADERS.length).getValues();
    const td = today_();
    rows.forEach(function (r) {
      if (!(r[0] instanceof Date)) return;
      if (Utilities.formatDate(r[0], tz_(), 'yyyy-MM-dd') !== td) return;
      const res = String(r[8]);
      ok += (res.match(/成功/g) || []).length;
      if (!/已重送成功/.test(res)) fail += (res.match(/失敗/g) || []).length;
    });
    logs = rows.slice(-60).reverse().map(function (r) {
      return {
        time: r[0] instanceof Date ? fmt_(r[0]) : String(r[0]),
        page: String(r[1]), platform: String(r[2]), type: String(r[3]), who: String(r[4]),
        text: String(r[5]), postId: String(r[6]), keyword: String(r[7]), result: String(r[8])
      };
    });
  }

  // 排隊數量
  const qsh = sheet_(SH_QUEUE, QUEUE_HEADERS);
  const qn = qsh.getLastRow();
  let waiting = 0;
  if (qn >= 2) {
    qsh.getRange(2, Q.status + 1, qn - 1, 1).getValues().forEach(function (r) { if (r[0] === '等待') waiting++; });
  }

  const rules = getRules_().map(function (r) {
    delete r.startMs; delete r.endMs;
    return r;
  });

  return {
    pages: pages,
    rules: rules,
    logs: logs,
    stats: { ok: ok, fail: fail, waiting: waiting },
    settings: {
      webhookUrl: ScriptApp.getService().getUrl() || '（請先部署成網頁應用程式）',
      verifyToken: prop_('VERIFY_TOKEN'),
      appId: prop_('APP_ID'),
      hasSecret: !!prop_('APP_SECRET'),
      lineTo: prop_('LINE_TO'),
      observe: isObserve_(),
      logAll: isLogAll_(),
      hasLineToken: !!prop_('LINE_TOKEN'),
      email: Session.getEffectiveUser().getEmail(),
      sheetUrl: ss_().getUrl(),
      logUrl: ss_().getUrl() + '#gid=' + sheet_(SH_LOG, LOG_HEADERS).getSheetId()
    }
  };
}

// 紀錄查詢：可依粉專、結果篩選
function logKind_(res) {
  res = String(res);
  if (/已重送成功|^已補發/.test(res)) return '觸發';
  if (/失敗/.test(res)) return '失敗';
  if (/觀察/.test(res)) return '觀察';
  if (/未觸發/.test(res)) return '未觸發';
  return '觸發';
}

function apiLogs(pw, opt) {
  auth_(pw);
  opt = opt || {};
  const sh = sheet_(SH_LOG, LOG_HEADERS);
  const n = sh.getLastRow();
  const out = { rows: [], counts: { 全部: 0, 觸發: 0, 觀察: 0, 未觸發: 0, 失敗: 0 }, today: { 觸發: 0, 觀察: 0, 未觸發: 0, 失敗: 0 } };
  if (n < 2) return out;
  const from = Math.max(2, n - 4999);
  const rows = sh.getRange(from, 1, n - from + 1, LOG_HEADERS.length).getValues();
  const td = today_();
  const limit = Math.min(Number(opt.limit) || 150, 500);
  for (let i = rows.length - 1; i >= 0; i--) {
    const r = rows[i];
    if (opt.page && String(r[1]) !== opt.page) continue;
    const k = logKind_(r[8]);
    out.counts.全部++;
    out.counts[k]++;
    if (r[0] instanceof Date && Utilities.formatDate(r[0], tz_(), 'yyyy-MM-dd') === td) out.today[k]++;
    if (opt.kind && opt.kind !== k) continue;
    if (out.rows.length < limit) {
      out.rows.push({
        time: r[0] instanceof Date ? fmt_(r[0]) : String(r[0]),
        page: String(r[1]), platform: String(r[2]), type: String(r[3]), who: String(r[4]),
        text: String(r[5]), postId: String(r[6]), keyword: String(r[7]), result: String(r[8]), kind: k
      });
    }
  }
  return out;
}

function apiImportPages(pw, userToken) {
  auth_(pw);
  ensureSetup_();
  let token = String(userToken || '').trim();
  if (!token) throw new Error('請貼上使用者存取權杖');

  let note;
  const appId = prop_('APP_ID'), secret = prop_('APP_SECRET');
  if (appId && secret) {
    const ex = graph_('get', 'oauth/access_token', {
      grant_type: 'fb_exchange_token', client_id: appId, client_secret: secret, fb_exchange_token: token
    });
    if (ex.ok && ex.data.access_token) {
      token = ex.data.access_token;
      note = '已自動換成長效權杖（粉專權杖不會過期）';
    } else {
      note = '長效權杖交換失敗：' + ex.msg;
    }
  } else {
    note = '注意：尚未在「設定」填 App ID 和 App Secret，這次匯入的權杖約 1 小時後失效';
  }

  const r = graph_('get', 'me/accounts', {
    fields: 'id,name,access_token,instagram_business_account{id,username}', limit: 100
  }, token);
  if (!r.ok) throw new Error('讀取粉專失敗 ' + r.msg);

  const pages = getPages_();
  const added = [];
  (r.data.data || []).forEach(function (a) {
    const ig = a.instagram_business_account || {};
    const obj = { pageId: a.id, name: a.name, token: a.access_token, igId: ig.id || '', igName: ig.username || '' };
    const i = pages.findIndex(function (p) { return p.pageId === a.id; });
    if (i >= 0) { obj.igOff = !!pages[i].igOff; pages[i] = obj; } else pages.push(obj);
    const s = subscribe_(obj);
    const igText = obj.igName ? '（IG：' + obj.igName + (obj.igOff ? '，已停用' : '') + '）' : '';
    added.push(a.name + igText + '　' + (s.ok ? '訂閱成功' : '訂閱失敗 ' + s.msg));
  });
  savePages_(pages);
  if (!added.length) note += '\n沒有讀到任何粉專，請確認權杖有勾選 pages_show_list 權限';
  return { note: note, added: added, data: loadData_() };
}

/* ---------- 規則 ---------- */
function apiSaveRule(pw, rule) {
  auth_(pw);
  const r = normRule_(rule);
  if (!r.pageId) throw new Error('請選擇粉專');
  const tps = typesOf_(r.type);
  if (!tps.length) throw new Error('請至少選一種：貼文留言、私訊或限動回覆');
  const useMode = tps.indexOf('留言') >= 0 && r.mode !== '任何留言';
  const useDmMode = (tps.indexOf('私訊') >= 0 || tps.indexOf('限動回覆') >= 0) && r.dmMode !== '任何留言';
  if ((useMode || useDmMode) && !splitKw_(r.keywords).length) throw new Error('請輸入關鍵字');
  r.type = tps.join(',');
  if ((tps.indexOf('私訊') >= 0 || tps.indexOf('限動回覆') >= 0) && !r.dm) throw new Error('有勾私訊或限動回覆，一定要填私訊內容');
  if (!r.publicReply && !r.dm) throw new Error('公開回覆和私訊內容至少要填一個');
  if ((r.links.length || r.quick.length) && !r.dm) throw new Error('有設定按鈕，就要填私訊內容');
  if (r.startMs && r.endMs && r.endMs <= r.startMs) throw new Error('結束時間要晚於開始時間');
  r.links.forEach(function (l) {
    if (!/^https?:\/\//i.test(l.url)) throw new Error('按鈕「' + l.title + '」的網址要以 http:// 或 https:// 開頭');
  });
  if (tps.indexOf('留言') < 0) { r.postId = ''; r.postText = ''; }

  const sh = sheet_(SH_RULES, RULE_HEADERS);
  const row = [
    txt_(r.id || Utilities.getUuid().slice(0, 8)), r.enabled ? 'Y' : 'N', txt_(r.pageId), r.platform, r.type,
    txt_(r.postId), txt_(r.postText), txt_(r.keywords), r.mode, txt_(r.publicReply), txt_(r.dm),
    txt_(r.exclude), r.topOnly ? 'Y' : 'N', txt_(r.startAt), txt_(r.endAt), r.delayMin, r.delayMax,
    txt_(JSON.stringify({ links: r.links, quick: r.quick })), new Date(), r.observe ? 'Y' : 'N', r.dmMode, r.once ? 'Y' : 'N'
  ];
  let done = false;
  if (r.id) {
    const n = sh.getLastRow();
    if (n >= 2) {
      const ids = sh.getRange(2, 1, n - 1, 1).getValues();
      for (let i = 0; i < ids.length; i++) {
        if (String(ids[i][0]) === r.id) {
          row[18] = sh.getRange(i + 2, 19).getValue() || new Date();
          sh.getRange(i + 2, 1, 1, RULE_HEADERS.length).setValues([row]);
          done = true;
          break;
        }
      }
    }
  }
  if (!done) sh.appendRow(row);
  refreshRulesCache_();
  return loadData_();
}

function apiTestRule(pw, rule, text, isReply) {
  auth_(pw);
  const r = normRule_(rule);
  r.enabled = true;
  // FB 貼文 ID 有底線（粉專ID_貼文ID），IG 貼文 ID 是純數字
  const firstPost = postsOf_(r.postId)[0] || '';
  const platform = r.platform !== '全部' ? r.platform : (firstPost && firstPost.indexOf('_') < 0 ? 'IG' : 'FB');
  const type = typesOf_(r.type)[0] || '留言';
  const m = matchRule_([r], {
    pageId: r.pageId, platform: platform, type: type, postId: firstPost, text: text, isReply: !!isReply
  }, true);
  if (!m) {
    if (type === '留言' && isReply && r.topOnly) return '不會觸發（這條規則只回第一層留言）';
    return '不會觸發';
  }

  let s = '會觸發（符合：' + m.hit + '）' + (r.observe ? '\n這條規則設定「只觀察」，實際不會送出' : '');
  const now = Date.now();
  if (r.startMs && now < r.startMs) s += '\n注意：還沒到開始時間，目前不會回覆';
  if (r.endMs && now > r.endMs) s += '\n注意：已經過了結束時間，目前不會回覆';
  if (r.delayMax > 0) s += '\n回覆前會隨機等待 ' + r.delayMin + '～' + r.delayMax + ' 秒';
  if (m.publicReply && type === '留言') s += '\n\n公開回覆：\n' + fill_(pick_(m.publicReply), '王小明');
  if (m.dm) {
    s += '\n\n私訊：\n' + fill_(pick_(m.dm), '王小明');
    if (m.links.length) s += '\n\n網址按鈕：' + m.links.map(function (l) { return '［' + l.title + '］'; }).join(' ');
    if (m.quick.length) s += '\n快速回覆按鈕：' + m.quick.map(function (q) { return '［' + q + '］'; }).join(' ');
  }
  return s;
}

// 失敗的訊息清單（最近 7 天，留言私訊超過 7 天 Meta 就不給送了）
function apiFailed(pw) {
  auth_(pw);
  const sh = sheet_(SH_QUEUE, QUEUE_HEADERS);
  const n = sh.getLastRow();
  const data = n >= 2 ? sh.getRange(2, 1, n - 1, QUEUE_HEADERS.length).getValues() : [];
  const pages = getPages_();
  const limit = Date.now() - 7 * 86400000;
  const out = [];
  for (let i = data.length - 1; i >= 0 && out.length < 50; i--) {
    const r = data[i];
    if (r[Q.status] !== '失敗') continue;
    const t = r[Q.added] instanceof Date ? r[Q.added].getTime() : 0;
    if (t && t < limit) continue;
    const pg = pages.find(function (p) { return p.pageId === String(r[Q.pageId]); });
    out.push({
      row: i + 2, time: t ? fmt_(r[Q.added]) : '', page: pg ? pg.name : String(r[Q.pageId]),
      platform: String(r[Q.platform]), action: String(r[Q.action]), who: String(r[Q.who]),
      text: String(r[Q.text]), result: String(r[Q.result])
    });
  }
  return out.concat(oldFailed_(data, pages, limit));
}

// 紀錄裡有失敗、但排隊表沒有資料的留言（舊版程式的失敗）→ 可以「找回留言再重送」
function oldFailed_(qdata, pages, limit) {
  const inQueue = {};
  qdata.forEach(function (r) { inQueue[String(r[Q.who]) + '|' + String(r[Q.text])] = 1; });
  const sh = sheet_(SH_LOG, LOG_HEADERS);
  const n = sh.getLastRow();
  if (n < 2) return [];
  const from = Math.max(2, n - 2999);
  const rows = sh.getRange(from, 1, n - from + 1, LOG_HEADERS.length).getValues();
  const out = [];
  for (let i = rows.length - 1; i >= 0 && out.length < 30; i--) {
    const r = rows[i];
    const t = r[0] instanceof Date ? r[0].getTime() : 0;
    if (!t || t < limit) continue;
    const res = String(r[8]);
    if (!/^留言/.test(String(r[3])) || !String(r[6])) continue;
    if (!/(公開|私訊):失敗/.test(res) || /已重送成功/.test(res)) continue;
    if (inQueue[String(r[4]) + '|' + String(r[5])]) continue;
    out.push({
      logRow: from + i, time: fmt_(r[0]), page: String(r[1]), platform: String(r[2]),
      action: /私訊:失敗/.test(res) ? '私訊' : '公開', who: String(r[4]), text: String(r[5]),
      result: res.replace(/^.*?((公開|私訊):失敗)/, '$1')
    });
  }
  return out;
}

// 找回原本的留言（用名字和內容在貼文底下找），再補送失敗的回覆
function apiResendLog(pw, logRow) {
  auth_(pw);
  const sh = sheet_(SH_LOG, LOG_HEADERS);
  logRow = Number(logRow);
  if (logRow < 2 || logRow > sh.getLastRow()) throw new Error('找不到這筆紀錄');
  const r = sh.getRange(logRow, 1, 1, LOG_HEADERS.length).getValues()[0];
  const res = String(r[8]);
  if (/已重送成功/.test(res)) return { data: loadData_(), failed: apiFailed(pw), msg: '這筆已經重送過了' };
  const page = getPages_().find(function (p) { return p.name === String(r[1]); });
  if (!page) throw new Error('找不到粉專「' + r[1] + '」');
  const platform = String(r[2]), who = String(r[4]), text = String(r[5]), postId = String(r[6]);

  // 1. 在貼文底下找這個人的這則留言
  const cid = findComment_(page, platform, postId, who, text, r[0] instanceof Date ? r[0].getTime() : 0);
  if (!cid) throw new Error('在貼文底下找不到「' + who + '」留的「' + text + '」，可能留言被刪除了，請手動私訊');

  // 2. 找出當時符合的規則，組出要回的內容
  const ev = { pageId: page.pageId, platform: platform, type: '留言', postId: postId, text: text, isReply: /回覆/.test(String(r[3])) };
  const rule = matchRule_(getRules_(), ev, true);
  if (!rule) throw new Error('現在已經沒有符合的規則（可能規則被改掉或停用），請手動回覆');

  // 3. 補送失敗的部分
  const out = [];
  let allOk = true;
  if (/私訊:失敗/.test(res)) {
    if (!rule.dm) throw new Error('規則沒有私訊內容');
    const x = send_(page, platform, '私訊留言', cid, { text: fill_(pick_(rule.dm), who), links: rule.links, quick: rule.quick });
    out.push('私訊:' + (x.ok ? '成功' : '失敗 ' + x.msg));
    if (!x.ok) allOk = false;
  }
  if (/公開:失敗/.test(res) && rule.publicReply) {
    const y = send_(page, platform, '公開', cid, { text: fill_(pick_(rule.publicReply), who) });
    out.push('公開:' + (y.ok ? '成功' : '失敗 ' + y.msg));
    if (!y.ok) allOk = false;
  }
  if (allOk) {
    const when = Utilities.formatDate(new Date(), tz_(), 'MM/dd HH:mm');
    sh.getRange(logRow, 9).setValue('已重送成功（' + when + '）｜原本：' + res);
  }
  log_([new Date(), page.name, platform, '補送', who, text, postId, rule.hit, out.join(' / ')]);
  return { data: loadData_(), failed: apiFailed(pw), msg: allOk ? '已重送成功' : '還是失敗：' + out.join(' / ') };
}

// 讀一篇貼文底下的留言（最多 1000 則）
function loadComments_(page, platform, postId) {
  const out = [];
  let params = platform === 'IG'
    ? { fields: 'id,text,username,timestamp,replies.limit(20){username}', limit: 100 }
    : { fields: 'id,message,from{id,name},created_time,can_reply_privately,comments.limit(20){from{id}}', limit: 100,
        filter: 'stream', order: 'reverse_chronological' };
  for (let pageNo = 0; pageNo < 15; pageNo++) {
    const g = graph_('get', postId + '/comments', params, page.token);
    if (!g.ok) throw new Error('讀取貼文留言失敗：' + g.msg);
    (g.data.data || []).forEach(function (c) {
      // 粉專已經在這則留言底下回過了嗎？
      let pageReplied = false;
      if (platform === 'IG') {
        pageReplied = ((c.replies && c.replies.data) || []).some(function (x) {
          return page.igUsername && x.username === page.igUsername;
        });
      } else {
        pageReplied = ((c.comments && c.comments.data) || []).some(function (x) {
          return x.from && String(x.from.id) === String(page.pageId);
        });
      }
      out.push({
        id: c.id,
        name: String((platform === 'IG' ? c.username : (c.from && c.from.name)) || ''),
        text: norm_(String((platform === 'IG' ? c.text : c.message) || '')).trim(),
        ms: Date.parse(String(c.created_time || c.timestamp || '').replace(/\+0000$/, 'Z')) || 0,
        pageReplied: pageReplied,
        // false 代表已經私訊過了（或超過 7 天），Meta 不讓再私訊
        dmDone: c.can_reply_privately === false
      });
    });
    const next = g.data.paging && g.data.paging.cursors && g.data.paging.next ? g.data.paging.cursors.after : '';
    if (!next) break;
    params = Object.assign({}, params, { after: next });
  }
  return out;
}

// 先用「名字 + 內容」找；Meta 沒給名字時，改用「內容 + 留言時間（紀錄時間前後 5 分鐘）」找
function matchComment_(list, who, text, logMs, used) {
  used = used || {};
  const want = norm_(text).trim();
  const same = list.filter(function (c) { return c.text === want && !used[c.id]; });
  const named = same.filter(function (c) { return c.name && c.name === who; });
  if (named.length) {
    // 同一人留了好幾次，取時間最接近的
    named.sort(function (a, b) { return Math.abs(a.ms - logMs) - Math.abs(b.ms - logMs); });
    return named[0].id;
  }
  if (!logMs) return '';
  const cands = same.filter(function (c) { return !c.name && c.ms && Math.abs(c.ms - logMs) <= 300000; })
    .map(function (c) { return { id: c.id, gap: Math.abs(c.ms - logMs) }; })
    .sort(function (a, b) { return a.gap - b.gap; });
  if (cands.length === 1 || (cands.length > 1 && cands[1].gap - cands[0].gap > 20000)) return cands[0].id;
  return '';
}

function matchCommentObj_(list, who, text, logMs, used) {
  const id = matchComment_(list, who, text, logMs, used);
  return id ? list.find(function (c) { return c.id === id; }) : null;
}

function findComment_(page, platform, postId, who, text, logMs) {
  return matchComment_(loadComments_(page, platform, postId), who, text, logMs);
}

/* =========================================================
 *  補發：規則改了之後，把以前沒回到（或只觀察）的留言補回覆
 * ========================================================= */
function backfillList_(rule) {
  const page = getPages_().find(function (p) { return p.pageId === rule.pageId; });
  if (!page) throw new Error('找不到這條規則的粉專');
  const sh = sheet_(SH_LOG, LOG_HEADERS);
  const n = sh.getLastRow();
  if (n < 2) return { page: page, list: [] };
  const from = Math.max(2, n - 4999);
  const rows = sh.getRange(from, 1, n - from + 1, LOG_HEADERS.length).getValues();
  const limit = Date.now() - 7 * 86400000;
  const r2 = Object.assign({}, rule, { enabled: true });

  // 已經回過的（同一人、同一篇貼文）不要重複
  const done = {};
  rows.forEach(function (r) {
    const res = String(r[8]);
    if (/^留言/.test(String(r[3])) && /公開:成功|私訊:成功|已補發|已重送成功/.test(res)) done[r[4] + '|' + r[6]] = 1;
    if (String(r[3]) === '補送' && /成功/.test(res)) done[r[4] + '|' + r[6]] = 1;
  });

  const seen = {};
  const list = [];
  rows.forEach(function (r, i) {
    const t = r[0] instanceof Date ? r[0].getTime() : 0;
    if (!t || t < limit) return;
    if (String(r[1]) !== page.name) return;
    const type = String(r[3]);
    if (!/^留言/.test(type) || !String(r[6])) return;
    const res = String(r[8]);
    if (!/^未觸發|觀察/.test(res) || /已補發/.test(res)) return;
    const key = r[4] + '|' + r[6];
    if (done[key] || seen[key]) return;
    const ev = { pageId: page.pageId, platform: String(r[2]), type: '留言', postId: String(r[6]),
      text: String(r[5]), isReply: /回覆/.test(type) };
    if (!matchRule_([r2], ev, true)) return;
    seen[key] = 1;
    list.push({ logRow: from + i, time: fmt_(r[0]), ms: t, platform: String(r[2]), who: String(r[4]),
      text: String(r[5]), postId: String(r[6]), isReply: ev.isReply, logCid: String(r[9] || '') });
  });

  // 到 Facebook 確認每一則留言的實際狀況：已經有人回過、已經私訊過、或留言被刪除的都排除
  const skip = { replied: 0, deleted: 0, unverified: 0, err: '' };
  const comments = {};
  const used = {};
  const final = [];
  list.forEach(function (x) {
    // 新紀錄有存留言編號 → 直接問 Facebook 這則留言的狀態，最準確
    if (x.logCid) {
      const st = commentState_(page, x.platform, x.logCid);
      if (st === 'gone') { skip.deleted++; return; }
      if (st === 'replied') { skip.replied++; return; }
      if (st === 'error') skip.unverified++;
      x.cid = x.logCid;
      final.push(x);
      return;
    }
    if (!comments[x.postId]) {
      try { comments[x.postId] = loadComments_(page, x.platform, x.postId); }
      catch (e) { comments[x.postId] = null; skip.err = String(e.message || e); }
    }
    const cl = comments[x.postId];
    if (!cl) { skip.unverified++; final.push(x); return; } // 讀不到就照紀錄判斷
    const c = matchCommentObj_(cl, x.who, x.text, x.ms, used);
    if (!c) { skip.unknown = (skip.unknown || 0) + 1; return; }
    used[c.id] = 1;
    if (c.pageReplied || c.dmDone) { skip.replied++; return; }
    x.cid = c.id;
    final.push(x);
  });
  return { page: page, list: final, skip: skip };
}

// 先看看有哪些可以補發
// 直接查一則留言：'ok' 可以補發、'replied' 已經回過、'gone' 被刪除、'error' 查不到
function commentState_(page, platform, cid) {
  const fields = platform === 'IG' ? 'id,replies.limit(20){username}' : 'id,can_reply_privately,comments.limit(20){from{id}}';
  const g = graph_('get', cid, { fields: fields }, page.token);
  if (!g.ok) return /does not exist|不存在|Unsupported get request/i.test(g.msg) ? 'gone' : 'error';
  const c = g.data;
  if (platform !== 'IG') {
    if (c.can_reply_privately === false) return 'replied';
    if (((c.comments && c.comments.data) || []).some(function (x) { return x.from && String(x.from.id) === String(page.pageId); })) return 'replied';
  } else if (page.igUsername && ((c.replies && c.replies.data) || []).some(function (x) { return x.username === page.igUsername; })) {
    return 'replied';
  }
  return 'ok';
}

function apiBackfillScan(pw, ruleId) {
  auth_(pw);
  const rule = getRules_().find(function (r) { return r.id === ruleId; });
  if (!rule) throw new Error('找不到這條規則');
  const b = backfillList_(rule);
  return {
    observe: rule.observe || isObserve_(),
    hasDm: !!rule.dm, hasPublic: !!rule.publicReply,
    skip: b.skip,
    list: b.list.map(function (x) { return { logRow: x.logRow, time: x.time, who: x.who, text: x.text }; })
  };
}

// 真的補發（一次最多 30 則，網頁會自動接著送下一批）
function apiBackfillSend(pw, ruleId, logRows) {
  auth_(pw);
  if (isObserve_()) throw new Error('設定裡的總開關是「全部只觀察」，要先改成「照各規則設定」才能補發');
  const rule = getRules_().find(function (r) { return r.id === ruleId; });
  if (!rule) throw new Error('找不到這條規則');
  if (rule.observe) throw new Error('這條規則是「只觀察」，要先改成「正式回覆」才能補發');
  const want = {};
  (logRows || []).slice(0, 30).forEach(function (x) { want[Number(x)] = 1; });
  const b = backfillList_(rule);
  const page = b.page;
  const items = b.list.filter(function (x) { return want[x.logRow]; });
  const sh = sheet_(SH_LOG, LOG_HEADERS);
  const comments = {};
  const used = {};
  const when = Utilities.formatDate(new Date(), tz_(), 'MM/dd HH:mm');
  let ok = 0, fail = 0, notFound = 0;
  const DAY7 = 7 * 86400000 - 3600000;

  items.forEach(function (x, idx) {
    // 每個人之間隨機隔 1～2.5 秒，避免一次大量相同回覆被 Facebook 當成垃圾訊息
    if (idx > 0) Utilities.sleep(1000 + Math.floor(Math.random() * 1500));
    if (!x.cid && !comments[x.postId]) {
      try { comments[x.postId] = loadComments_(page, x.platform, x.postId); }
      catch (e) { comments[x.postId] = []; }
    }
    const cid = x.cid || matchComment_(comments[x.postId], x.who, x.text, x.ms, used);
    if (!cid) {
      notFound++;
      log_([new Date(), page.name, x.platform, '補發', x.who, x.text, x.postId, '', '補發失敗：找不到這則留言（可能被刪除）']);
      return;
    }
    used[cid] = 1;
    const info = { page: page, platform: x.platform, who: x.who, text: x.text, keyword: '' };
    const jobs = [];
    if (rule.publicReply) jobs.push(['公開', { text: fill_(pick_(rule.publicReply), x.who) }]);
    // 留言私訊只能在 7 天內送
    if (rule.dm && Date.now() - x.ms < DAY7) {
      jobs.push(['私訊留言', { text: fill_(pick_(rule.dm), x.who), links: rule.links, quick: rule.quick }]);
    }
    const out = [];
    let allOk = true;
    jobs.forEach(function (j) {
      let r = send_(page, x.platform, j[0], cid, j[1]);
      if (!r.ok && r.transient) { Utilities.sleep(1500); r = send_(page, x.platform, j[0], cid, j[1]); }
      if (r.ok) out.push((j[0] === '公開' ? '公開:' : '私訊:') + '成功' + (r.note || ''));
      else {
        allOk = false;
        out.push((j[0] === '公開' ? '公開:' : '私訊:') + '失敗 ' + r.msg);
        try { enqueue_(info, j[0], cid, j[1], new Date(), r.msg); } catch (e) {}
      }
    });
    if (rule.dm && Date.now() - x.ms >= DAY7) out.push('私訊:超過 7 天，Meta 不給送，只補公開回覆');
    if (allOk) ok++; else fail++;
    const old = String(sh.getRange(x.logRow, 9).getValue());
    sh.getRange(x.logRow, 9).setValue('已補發（' + when + '）｜原本：' + old);
    log_([new Date(), page.name, x.platform, '補發', x.who, x.text, x.postId, '', out.join(' / ')]);
  });
  return { ok: ok, fail: fail, notFound: notFound, done: items.length };
}

// 重送：把指定的失敗訊息改回「等待」，馬上處理
function apiResend(pw, rows) {
  auth_(pw);
  const sh = sheet_(SH_QUEUE, QUEUE_HEADERS);
  (rows || []).forEach(function (row) {
    row = Number(row);
    if (row < 2 || row > sh.getLastRow()) return;
    if (sh.getRange(row, Q.status + 1).getValue() !== '失敗') return;
    sh.getRange(row, Q.due + 1).setValue(new Date());
    sh.getRange(row, Q.status + 1, 1, 3).setValues([['等待', '手動重送', 0]]);
  });
  setProp_('QUEUE_HAS', '1');
  processQueue();
  return { data: loadData_(), failed: apiFailed(pw) };
}

function apiProcessQueueNow(pw) {
  auth_(pw);
  setProp_('QUEUE_HAS', '1');
  processQueue();
  return loadData_();
}
